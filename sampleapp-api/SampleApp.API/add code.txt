import { useState, type ChangeEvent } from 'react';
import {
  Alert,
  Box,
  Button,
  Paper,
  TextField,
  Typography,
} from '@mui/material';
import { Send } from 'lucide-react';

type MicropostComposerProps = {
  onSubmit: (content: string) => Promise<void>;
  loading?: boolean;
};

export const MicropostComposer = ({
  onSubmit,
  loading = false,
}: MicropostComposerProps) => {
  const [content, setContent] = useState('');
  const [error, setError] = useState('');

  const handleChange = (e: ChangeEvent<HTMLInputElement>) => {
    setContent(e.target.value);

    if (error) {
      setError('');
    }
  };

  const handleSubmit = async () => {
    if (!content.trim()) {
      setError('Введите текст сообщения');
      return;
    }

    if (content.trim().length < 2) {
      setError('Минимум 2 символа');
      return;
    }

    try {
      await onSubmit(content.trim());
      setContent('');
      setError('');
    } catch (err: any) {
      setError(err?.message || 'Не удалось создать сообщение');
    }
  };

  return (
    <Paper sx={{ p: 3, mb: 3 }}>
      <Typography variant="h6" gutterBottom>
        Новое сообщение
      </Typography>

      {error && (
        <Alert severity="error" sx={{ mb: 2 }}>
          {error}
        </Alert>
      )}

      <Box display="flex" flexDirection="column" gap={2}>
        <TextField
          multiline
          minRows={3}
          maxRows={8}
          fullWidth
          placeholder="Что у вас нового?"
          value={content}
          onChange={handleChange}
          disabled={loading}
        />

        <Box display="flex" justifyContent="space-between" alignItems="center">
          <Typography variant="caption" color="text.secondary">
            {content.length} символов
          </Typography>

          <Button
            variant="contained"
            startIcon={<Send size={18} />}
            onClick={handleSubmit}
            disabled={loading}
          >
            {loading ? 'Публикация...' : 'Опубликовать'}
          </Button>
        </Box>
      </Box>
    </Paper>
  );
};

import { useState } from 'react';
import {
  Alert,
  Avatar,
  Box,
  Button,
  Card,
  CardContent,
  IconButton,
  Stack,
  TextField,
  Tooltip,
  Typography,
} from '@mui/material';
import {
  Edit2,
  Save,
  Trash2,
  X,
  User as UserIcon,
} from 'lucide-react';
import { useNavigate } from 'react-router-dom';
import type { Micropost } from '../types/swagger';
import { useAuth } from '../contexts/AuthContext';

type MicropostCardProps = {
  post: Micropost;
  onUpdate: (id: number, content: string) => Promise<void>;
  onDelete: (id: number) => Promise<void>;
};

export const MicropostCard = ({
  post,
  onUpdate,
  onDelete,
}: MicropostCardProps) => {
  const navigate = useNavigate();
  const { user } = useAuth();

  const [isEditing, setIsEditing] = useState(false);
  const [content, setContent] = useState(post.content || '');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const isOwnPost = user?.id === post.userId;

  const handleSave = async () => {
    if (!content.trim()) {
      setError('Сообщение не может быть пустым');
      return;
    }

    try {
      setLoading(true);
      setError('');
      await onUpdate(post.id, content.trim());
      setIsEditing(false);
    } catch (err: any) {
      setError(err?.message || 'Не удалось обновить сообщение');
    } finally {
      setLoading(false);
    }
  };

  const handleDelete = async () => {
    const confirmed = window.confirm('Удалить сообщение?');
    if (!confirmed) return;

    try {
      setLoading(true);
      setError('');
      await onDelete(post.id);
    } catch (err: any) {
      setError(err?.message || 'Не удалось удалить сообщение');
      setLoading(false);
    }
  };

  const authorName = post.user?.name || post.user?.login || `User #${post.userId}`;
  const authorLogin = post.user?.login || `user${post.userId}`;
  const createdAt = new Date(post.createdAt).toLocaleString('ru-RU');

  return (
    <Card variant="outlined">
      <CardContent>
        <Box display="flex" justifyContent="space-between" alignItems="flex-start" gap={2}>
          <Box display="flex" gap={2} flex={1}>
            <Avatar
              sx={{ cursor: 'pointer' }}
              onClick={() => navigate(`/profile/${post.userId}`)}
            >
              {authorLogin.charAt(0).toUpperCase()}
            </Avatar>

            <Box flex={1}>
              <Box display="flex" alignItems="center" gap={1} flexWrap="wrap" mb={1}>
                <Typography
                  variant="subtitle1"
                  sx={{ fontWeight: 600, cursor: 'pointer' }}
                  onClick={() => navigate(`/profile/${post.userId}`)}
                >
                  {authorName}
                </Typography>

                <Typography variant="body2" color="text.secondary">
                  @{authorLogin}
                </Typography>

                <Typography variant="body2" color="text.secondary">
                  •
                </Typography>

                <Typography variant="body2" color="text.secondary">
                  {createdAt}
                </Typography>
              </Box>

              {error && (
                <Alert severity="error" sx={{ mb: 2 }}>
                  {error}
                </Alert>
              )}

              {isEditing ? (
                <Stack spacing={2}>
                  <TextField
                    multiline
                    minRows={3}
                    fullWidth
                    value={content}
                    onChange={(e) => setContent(e.target.value)}
                    disabled={loading}
                  />

                  <Box display="flex" gap={1}>
                    <Button
                      variant="contained"
                      startIcon={<Save size={16} />}
                      onClick={handleSave}
                      disabled={loading}
                    >
                      Сохранить
                    </Button>

                    <Button
                      variant="outlined"
                      startIcon={<X size={16} />}
                      onClick={() => {
                        setIsEditing(false);
                        setContent(post.content || '');
                        setError('');
                      }}
                      disabled={loading}
                    >
                      Отмена
                    </Button>
                  </Box>
                </Stack>
              ) : (
                <Typography variant="body1" sx={{ whiteSpace: 'pre-wrap' }}>
                  {post.content || '—'}
                </Typography>
              )}
            </Box>
          </Box>

          {isOwnPost && !isEditing && (
            <Box display="flex" gap={1}>
              <Tooltip title="Редактировать">
                <IconButton onClick={() => setIsEditing(true)} disabled={loading}>
                  <Edit2 size={18} />
                </IconButton>
              </Tooltip>

              <Tooltip title="Удалить">
                <IconButton onClick={handleDelete} disabled={loading} color="error">
                  <Trash2 size={18} />
                </IconButton>
              </Tooltip>
            </Box>
          )}
        </Box>
      </CardContent>
    </Card>
  );
};

src/hooks/useMicroposts.ts

import { useEffect, useState } from 'react';
import {
  createMicropost,
  deleteMicropost,
  getMicroposts,
  updateMicropost,
} from '../api/microposts';
import type { Micropost } from '../types/swagger';
import { useAuth } from '../contexts/AuthContext';
import { useLoading } from '../contexts/LoadingContext';

export const useMicroposts = () => {
  const [posts, setPosts] = useState<Micropost[]>([]);
  const [error, setError] = useState<string | null>(null);
  const { user } = useAuth();
  const { withLoading } = useLoading();

  const loadPosts = async () => {
    try {
      setError(null);
      const data = await withLoading(getMicroposts());
      const sorted = [...data].sort(
        (a, b) =>
          new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
      );
      setPosts(sorted);
    } catch (err) {
      setError('Не удалось загрузить ленту сообщений');
      console.error(err);
    }
  };

  useEffect(() => {
    loadPosts();
  }, []);

  const createPost = async (content: string) => {
    if (!user?.id) {
      throw new Error('Нужно войти в систему');
    }

    const created = await withLoading(
      createMicropost({
        content,
        userId: user.id,
      })
    );

    setPosts((prev) =>
      [created, ...prev].sort(
        (a, b) =>
          new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
      )
    );
  };

  const editPost = async (id: number, content: string) => {
    const updated = await withLoading(updateMicropost(id, { content }));

    setPosts((prev) =>
      prev.map((post) => (post.id === id ? { ...post, ...updated } : post))
    );
  };

  const removePost = async (id: number) => {
    await withLoading(deleteMicropost(id));
    setPosts((prev) => prev.filter((post) => post.id !== id));
  };

  return {
    posts,
    error,
    refetch: loadPosts,
    createPost,
    editPost,
    removePost,
  };
};

import {
  Alert,
  Box,
  Button,
  Chip,
  Container,
  Paper,
  Typography,
} from '@mui/material';
import { MessageSquare, RefreshCw } from 'lucide-react';
import { useMicroposts } from '../hooks/useMicroposts';
import { useLoading } from '../contexts/LoadingContext';
import { MicropostComposer } from '../components/MicropostComposer';
import { MicropostCard } from '../components/MicropostCard';
import { ButtonLoader } from '../components/ButtonLoader';

export const FeedPage = () => {
  const { posts, error, refetch, createPost, editPost, removePost } =
    useMicroposts();
  const { isLoading } = useLoading();

  return (
    <Container
      maxWidth={false}
      sx={{ py: 4, px: { xs: 2, sm: 3, md: 4, lg: 6 } }}
    >
      <Box sx={{ width: '100%' }}>
        <Paper sx={{ p: 3, mb: 3 }}>
          <Box
            display="flex"
            justifyContent="space-between"
            alignItems="center"
            gap={2}
            flexWrap="wrap"
          >
            <Box display="flex" alignItems="center" gap={2}>
              <MessageSquare size={28} color="#1976d2" />
              <Typography variant="h4">Лента сообщений</Typography>
              <Chip
                label={`Всего: ${posts.length}`}
                size="small"
                color="primary"
                variant="outlined"
              />
            </Box>

            <Button
              variant="contained"
              onClick={refetch}
              disabled={isLoading}
              startIcon={isLoading ? <ButtonLoader /> : <RefreshCw size={18} />}
            >
              {isLoading ? 'Загрузка...' : 'Обновить'}
            </Button>
          </Box>
        </Paper>

        <MicropostComposer onSubmit={createPost} loading={isLoading} />

        {error && (
          <Alert severity="error" sx={{ mb: 3 }}>
            {error}
          </Alert>
        )}

        <Box display="flex" flexDirection="column" gap={2}>
          {posts.length === 0 ? (
            <Paper sx={{ p: 4, textAlign: 'center' }}>
              <Typography color="text.secondary">
                Сообщений пока нет. Опубликуйте первое.
              </Typography>
            </Paper>
          ) : (
            posts.map((post) => (
              <MicropostCard
                key={post.id}
                post={post}
                onUpdate={editPost}
                onDelete={removePost}
              />
            ))
          )}
        </Box>
      </Box>
    </Container>
  );
};

import { useEffect } from 'react';
import { BrowserRouter, Navigate, Route, Routes } from 'react-router-dom';
import { ThemeProvider, createTheme } from '@mui/material/styles';
import CssBaseline from '@mui/material/CssBaseline';
import { AuthProvider } from './contexts/AuthContext';
import { LoadingProvider } from './contexts/LoadingContext';
import { GlobalLoader } from './components/GlobalLoader';
import { Header } from './components/Header';
import { ProtectedRoute } from './components/ProtectedRoute';
import { HomePage } from './pages/HomePage';
import { UsersPage } from './pages/UsersPage';
import { ProfilePage } from './pages/ProfilePage';
import { EditProfilePage } from './pages/EditProfilePage';
import { FeedPage } from './pages/FeedPage';
import { LoginPage } from './pages/LoginPage';
import { RegisterPage } from './pages/RegisterPage';
import { LoadingDemoPage } from './pages/LoadingDemoPage';
import { GuardsDemoPage } from './pages/GuardsDemoPage';
import { TableDemoPage } from './pages/TableDemoPage';
import { NotFoundPage } from './pages/NotFoundPage';
import { ServerErrorPage } from './pages/ServerErrorPage';
import { setLoadingCallback } from './api/client';
import { useLoading } from './contexts/LoadingContext';

const theme = createTheme({
  palette: {
    primary: { main: '#1976d2' },
  },
});

const LoadingBridge = () => {
  const { setLoading } = useLoading();

  useEffect(() => {
    setLoadingCallback(setLoading);
  }, [setLoading]);

  return null;
};

function App() {
  return (
    <ThemeProvider theme={theme}>
      <CssBaseline />
      <LoadingProvider>
        <AuthProvider>
          <BrowserRouter>
            <LoadingBridge />
            <GlobalLoader message="Загрузка..." />
            <Header />

            <Routes>
              <Route path="/" element={<HomePage />} />
              <Route path="/login" element={<LoginPage />} />
              <Route path="/register" element={<RegisterPage />} />
              <Route path="/loading-demo" element={<LoadingDemoPage />} />
              <Route path="/guards-demo" element={<GuardsDemoPage />} />
              <Route path="/table-demo" element={<TableDemoPage />} />

              <Route
                path="/feed"
                element={
                  <ProtectedRoute>
                    <FeedPage />
                  </ProtectedRoute>
                }
              />

              <Route
                path="/users"
                element={
                  <ProtectedRoute>
                    <UsersPage />
                  </ProtectedRoute>
                }
              />

              <Route
                path="/profile/:id"
                element={
                  <ProtectedRoute>
                    <ProfilePage />
                  </ProtectedRoute>
                }
              />

              <Route
                path="/profile/:id/edit"
                element={
                  <ProtectedRoute>
                    <EditProfilePage />
                  </ProtectedRoute>
                }
              />

              <Route path="/404" element={<NotFoundPage />} />
              <Route path="/500" element={<ServerErrorPage />} />
              <Route path="*" element={<Navigate to="/404" replace />} />
            </Routes>
          </BrowserRouter>
        </AuthProvider>
      </LoadingProvider>
    </ThemeProvider>
  );
}

export default App;
