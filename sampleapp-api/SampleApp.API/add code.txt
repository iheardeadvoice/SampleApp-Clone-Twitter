public class UserUpdateDto
{
    public string Name { get; set; } = string.Empty;
    public string Login { get; set; } = string.Empty;
}

2) В файл

Если он уже есть — второй раз не добавляй.

3) В этот же UsersController.cs вставь новый метод внутрь класса UsersController

Вставь его после GetUserById или в любое место внутри класса:

[Authorize]
[HttpPut("{id}")]
[SwaggerOperation(Summary = "Обновление пользователя", OperationId = "UpdateUser")]
[SwaggerResponse(200, "Пользователь обновлен", typeof(UserDto))]
[SwaggerResponse(404, "Пользователь не найден")]
public async Task<IActionResult> UpdateUser(int id, [FromBody] UserUpdateDto updateDto)
{
    try
    {
        var user = await _repo.FindUserByIdAsync(id);

        user.Name = updateDto.Name;
        user.Login = updateDto.Login;

        await _repo.UpdateUserAsync(user);

        return Ok(user.ToDto());
    }
    catch (NotFoundException)
    {
        return NotFound($"Пользователь с id={id} не найден");
    }
}


⸻
